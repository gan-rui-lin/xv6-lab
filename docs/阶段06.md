## 一、实验基本信息
### （一）实验目的
1. 完善进程结构体定义与核心系统调用（fork、exit、wait等），实现进程完整生命周期管理。
2. 基于时间片轮转算法实现进程调度机制，支持多进程上下文切换与CPU资源分配。
3. 实现sleep/wakeup进程同步原语，解决进程等待与唤醒的协调问题，避免丢失唤醒等异常。
4. 深入理解进程状态转换逻辑、调度时机及多核环境下的进程管理优化思路。

### （二）实验环境
- 操作系统：xv6
- 架构：RISC-V
- 开发工具：GCC、QEMU（模拟器）、GDB（调试）、汇编器

## 二、实验核心原理
### （一）进程结构体与状态定义
1. 进程结构体扩展：在原有基础上新增自旋锁、父进程指针、退出状态、睡眠等待对象等字段，支撑多进程管理与同步。
2. 进程状态枚举：定义UNUSED（未使用）、RUNNABLE（就绪）、RUNNING（运行）、SLEEPING（睡眠）、ZOMBIE（僵尸）五种状态，覆盖完整生命周期。

### （二）时间片轮转调度算法
核心逻辑：为每个进程分配固定时间片，时钟中断触发时递减当前进程时间片，时间片为0时触发调度，切换至下一个就绪进程，同时重置当前进程时间片，保证进程公平占用CPU。

### （三）进程同步机制
sleep/wakeup机制：进程等待资源时调用sleep进入睡眠状态并释放CPU，资源就绪时通过wakeup唤醒对应进程，切换为就绪态等待调度，避免忙等待导致的CPU浪费。

## 三、实验步骤与关键代码
### （一）任务1：进程核心函数实现
#### 1. 进程初始化与资源管理
```c
// 全局变量定义
static proc_t procs[NPROC];
static proc_t* proczero;
static int global_pid = 1;
static spinlock_t lk_pid;

// 进程初始化函数
void proc_init(void) {
    initlock(&lk_pid, "global_pid");
    // 初始化进程数组，设置内核栈
    for (int i = 0; i < NPROC; i++) {
        procs[i].state = UNUSED;
        procs[i].kstack = KSTACK((uint64)&procs[i]);
        proc_free(&procs[i]);
    }
    // 创建首个进程proczero
    proczero = proc_alloc();
    proczero->pid = 0;
    proczero->state = RUNNABLE;
}

// 进程分配函数
proc_t* proc_alloc(void) {
    proc_t *p;
    acquire(&lk_pid);
    // 查找未使用的进程槽位
    for (p = procs; p < &procs[NPROC]; p++) {
        if (p->state == UNUSED) {
            // 初始化进程字段
            p->state = USED;
            p->pid = global_pid++;
            initlock(&p->lk, "proc");
            // 分配页表与内核栈
            p->pgtbl = proc_pgtbl_init((uint64)p->tf);
            p->tf = (trapframe_t*)pmem_alloc(PGSIZE);
            memset(p->tf, 0, PGSIZE);
            release(&lk_pid);
            return p;
        }
    }
    release(&lk_pid);
    return NULL; // 进程表满
}

// 进程资源释放函数
void proc_free(proc_t *p) {
    acquire(&p->lk);
    // 释放页表、陷阱帧等资源
    pgtbl_free(p->pgtbl);
    pmem_free((uint64)p->tf, PGSIZE);
    // 重置进程状态与字段
    p->state = UNUSED;
    p->parent = NULL;
    p->exit_state = 0;
    p->sleep_space = NULL;
    release(&p->lk);
}
```

#### 2. fork/exit/wait系统调用实现
```c
// fork系统调用：创建子进程
int sys_fork(void) {
    proc_t *parent = myproc();
    proc_t *child = proc_alloc();
    if (child == NULL) return -1;

    // 设置父进程与子进程关系
    child->parent = parent;
    // 复制父进程用户内存空间与陷阱帧
    copyproc(parent, child);
    // 子进程返回值设为0，父进程返回子进程PID
    child->tf->a0 = 0;
    // 复制打开的文件描述符
    for (int i = 0; i < NOFILE; i++) {
        child->ofile[i] = parent->ofile[i];
    }
    // 子进程设为就绪态
    child->state = RUNNABLE;
    return child->pid;
}

// exit系统调用：进程退出
void sys_exit(int status) {
    proc_t *p = myproc();
    acquire(&p->lk);
    // 设置退出状态，切换为僵尸态
    p->exit_state = status;
    p->state = ZOMBIE;
    // 托付子进程给proczero
    proc_reparent(p);
    // 唤醒父进程
    proc_wakeup_one(p->parent);
    release(&p->lk);
    // 放弃CPU，触发调度
    proc_yield();
}

// wait系统调用：等待子进程退出
int sys_wait(int *status) {
    proc_t *parent = myproc();
    while (1) {
        int found = 0;
        // 遍历查找子进程
        for (proc_t *p = procs; p < &procs[NPROC]; p++) {
            acquire(&p->lk);
            if (p->parent == parent && p->state == ZOMBIE) {
                // 读取退出状态
                if (status != NULL) {
                    *status = p->exit_state;
                }
                // 释放僵尸进程资源
                proc_free(p);
                found = 1;
            }
            release(&p->lk);
            if (found) return 0;
        }
        // 未找到僵尸进程，进入睡眠
        proc_sleep(parent, &parent->lk);
    }
}

// 进程托付函数：父进程退出时托付子进程给proczero
void proc_reparent(proc_t *p) {
    for (proc_t *child = procs; child < &procs[NPROC]; child++) {
        if (child->parent == p) {
            child->parent = proczero;
        }
    }
}
```

#### 2. 调度器核心函数
```c
#define TIME_SLICE 10  // 时间片大小（时钟中断次数）

// 调度触发函数：保存上下文并切换到调度器
void proc_sched(void) {
    proc_t *p = myproc();
    cpu_t *c = mycpu();
    // 检查锁状态与中断状态
    if (holding(&p->lk) || !intr_get()) {
        panic("proc_sched: invalid state");
    }
    // 保存当前进程状态为就绪态
    p->state = RUNNABLE;
    // 切换到CPU调度器上下文
    swtch(&p->ctx, &c->ctx);
    // 恢复中断状态
    intr_on();
}

// 调度器主函数：时间片轮转选择就绪进程
void proc_scheduler(void) {
    cpu_t *c = mycpu();
    c->proc = NULL;
    proc_t *p;
    for (;;) {
        intr_on();  // 开启中断，允许时钟中断触发调度
        for (p = procs; p < &procs[NPROC]; p++) {
            acquire(&p->lk);
            if (p->state == RUNNABLE) {
                // 切换进程状态为运行态
                p->state = RUNNING;
                c->proc = p;
                // 重置时间片
                p->time_slice = TIME_SLICE;
                // 切换上下文到目标进程
                swtch(&c->ctx, &p->ctx);
                // 调度返回后重置当前进程
                c->proc = NULL;
            }
            release(&p->lk);
        }
    }
}

// 进程放弃CPU：主动触发调度
void proc_yield(void) {
    acquire(&myproc()->lk);
    proc_sched();
    release(&myproc()->lk);
}

// 时钟中断处理：递减时间片，触发调度
void timer_handler(void) {
    proc_t *p = myproc();
    if (p == NULL) return;
    // 时间片递减，为0时触发调度
    if (--p->time_slice <= 0) {
        proc_yield();
    }
    // 唤醒超时进程
    timer_update();
}
```

### （三）任务3：sleep/wakeup同步原语实现
```c
// 进程睡眠函数：等待指定对象
void proc_sleep(void *chan, spinlock_t *lk) {
    proc_t *p = myproc();
    // 释放传入的锁，避免死锁
    release(lk);
    acquire(&p->lk);
    // 设置睡眠对象，切换状态为睡眠态
    p->sleep_space = chan;
    p->state = SLEEPING;
    // 触发调度，释放CPU
    proc_sched();
    // 唤醒后重新获取锁
    acquire(lk);
    release(&p->lk);
}

// 唤醒单个进程：仅唤醒指定父进程
void proc_wakeup_one(proc_t *parent) {
    acquire(&parent->lk);
    if (parent->state == SLEEPING) {
        parent->state = RUNNABLE;
    }
    release(&parent->lk);
}

// 唤醒所有等待对象的进程
void proc_wakeup(void *chan) {
    for (proc_t *p = procs; p < &procs[NPROC]; p++) {
        acquire(&p->lk);
        if (p->state == SLEEPING && p->sleep_space == chan) {
            p->state = RUNNABLE;
            p->sleep_space = NULL;
        }
        release(&p->lk);
    }
}
```

## 四、实验关键问题解答
### （一）进程管理核心问题
1. 父进程退出子进程未退出的处理：通过proc_reparent函数将子进程托付给根进程proczero，由于proczero作为进程树根部不会退出，确保子进程始终有父进程可调用wait回收资源，避免孤儿进程资源泄漏。

2. wait函数避免忙等待的方案：将原循环中的proc_yield替换为proc_sleep，父进程未找到僵尸子进程时进入睡眠态，释放CPU给其他进程；子进程退出时通过proc_wakeup_one主动唤醒父进程，大幅降低CPU占用。

3. 进程状态转换触发条件：
- UNUSED→USED：proc_alloc分配进程槽位时。
- USED→RUNNABLE：进程初始化完成、sleep唤醒、时间片到调度时。
- RUNNABLE→RUNNING：调度器选中进程并切换上下文时。
- RUNNING→SLEEPING：调用proc_sleep等待资源时。
- RUNNING→ZOMBIE：调用sys_exit退出时。
- ZOMBIE→UNUSED：父进程调用wait回收资源时。

### （二）调度机制问题
1. 时钟中断触发调度的原因：时钟中断是系统周期性触发的事件，可确保每个进程占用CPU的时间不超过设定的时间片，实现公平的时间片轮转调度，避免单个进程长期占用CPU。

2. 轮转调度的公平性：轮转调度为每个进程分配相同的时间片，按顺序轮流执行，确保所有就绪进程获得平等的CPU使用权，无优先级差异时公平性较高，但未考虑进程优先级与紧急程度，实时性较差。

3. 上下文切换保存寄存器的选择：仅保存被调用者保存寄存器（s0-s11、ra、sp），原因是调用者保存寄存器由函数调用者负责保存，上下文切换过程中无需内核额外保存，可减少切换开销；被调用者保存寄存器需由被调用者保存，内核切换时必须主动保存以保证进程恢复后执行一致性。

### （三）同步机制问题
1. sleep/wakeup避免丢失唤醒的锁规则：sleep前释放用户传入的锁，唤醒后重新获取；wakeup操作需持有被唤醒进程的锁，确保进程睡眠状态的修改与唤醒操作原子性，避免进程在睡眠状态切换过程中丢失唤醒信号。

2. 自旋锁与sleep的区别：自旋锁获取不到资源时CPU空转等待，适用于短时间等待场景；sleep获取不到资源时进程进入睡眠态并释放CPU，适用于长时间等待场景，可提高CPU利用率。

### （四）性能与扩展性问题
1. fork性能瓶颈解决：fork的核心瓶颈是复制父进程内存空间，可采用写时复制（Copy-On-Write）优化，初始时父子进程共享物理内存，仅当某一方修改内存时才复制对应页面，减少内存拷贝开销。

2. 上下文切换开销降低方案：减少上下文切换时保存的寄存器数量（仅保留必要的被调用者保存寄存器）、优化swtch汇编指令执行效率、合并频繁的调度操作，降低切换频率。

3. 多核调度与负载均衡：多核环境下为每个CPU配备独立调度器，维护本地就绪进程队列；定期检查各CPU队列的进程数量，将进程从负载高的CPU迁移到负载低的CPU，实现负载均衡。
